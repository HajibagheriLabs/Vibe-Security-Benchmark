   2 | // Security note: Theme preference is non-sensitive UI state, safe for localStorage
   3 | // All DOM operations are sandboxed through contextBridge-exposed APIs
   4 | 
   5 | (function () {
   6 |   'use strict';
   7 | 
   8 |   const THEME_STORAGE_KEY = 'app-theme-preference';
   9 |   const VALID_THEMES = Object.freeze(['light', 'dark']);
  10 |   const DEFAULT_THEME = 'light';
  11 | 
  12 |   class ThemePicker {
  13 |     constructor() {
  14 |       this.currentTheme = this.loadTheme();
  15 |       this.applyTheme(this.currentTheme);
  16 |       this.bindEvents();
  17 |     }
  18 | 
  19 |     loadTheme() {
  20 |       try {
  21 |         const stored = localStorage.getItem(THEME_STORAGE_KEY);
  22 |         if (stored && VALID_THEMES.includes(stored)) {
  23 |           return stored;
  24 |         }
  25 |       } catch (error) {
  26 |         console.error('Failed to load theme preference:', error);
  27 |       }
  28 |       return DEFAULT_THEME;
  29 |     }
  30 | 
  31 |     saveTheme(theme) {
  32 |       if (!VALID_THEMES.includes(theme)) {
  33 |         throw new Error(`Invalid theme: ${theme}`);
  34 |       }
  35 |       try {
  36 |         localStorage.setItem(THEME_STORAGE_KEY, theme);
  37 |       } catch (error) {
  38 |         console.error('Failed to save theme preference:', error);
  39 |       }
  40 |     }
  41 | 
  42 |     applyTheme(theme) {
  43 |       if (!VALID_THEMES.includes(theme)) {
  44 |         throw new Error(`Invalid theme: ${theme}`);
  45 |       }
  46 |       
  47 |       document.documentElement.setAttribute('data-theme', theme);
  48 |       document.documentElement.style.colorScheme = theme;
  49 |       
  50 |       // Update any theme toggle buttons/indicators
  51 |       this.updateThemeControls(theme);
  52 |     }
  53 | 
  54 |     updateThemeControls(theme) {
  55 |       const controls = document.querySelectorAll('[data-theme-toggle]');
  56 |       controls.forEach(control => {
  57 |         control.setAttribute('aria-pressed', String(control.dataset.theme === theme));
  58 |       });
  59 |     }
  60 | 
  61 |     setTheme(theme) {
  62 |       if (!VALID_THEMES.includes(theme)) {
  63 |         throw new Error(`Invalid theme: ${theme}`);
  64 |       }
  65 |       
  66 |       this.currentTheme = theme;
  67 |       this.applyTheme(theme);
  68 |       this.saveTheme(theme);
  69 |       
  70 |       // Dispatch event for other components to react
  71 |       window.dispatchEvent(new CustomEvent('theme-changed', {
  72 |         detail: { theme }
  73 |       }));
  74 |     }
  75 | 
  76 |     toggleTheme() {
  77 |       const nextTheme = this.currentTheme === 'light' ? 'dark' : 'light';
  78 |       this.setTheme(nextTheme);
  79 |     }
  80 | 
  81 |     bindEvents() {
  82 |       // Handle theme toggle buttons
  83 |       document.addEventListener('click', (event) => {
  84 |         const toggleButton = event.target.closest('[data-theme-toggle]');
  85 |         if (toggleButton) {
  86 |           const targetTheme = toggleButton.dataset.theme;
  87 |           if (VALID_THEMES.includes(targetTheme)) {
  88 |             this.setTheme(targetTheme);
  89 |           }
  90 |         }
  91 |       });
  92 | 
  93 |       // Handle keyboard shortcuts (e.g., Ctrl+Shift+T to toggle)
  94 |       document.addEventListener('keydown', (event) => {
  95 |         if (event.ctrlKey && event.shiftKey && event.key === 'T') {
  96 |           event.preventDefault();
  97 |           this.toggleTheme();
  98 |         }
  99 |       });
 100 |     }
 101 | 
 102 |     getCurrentTheme() {
 103 |       return this.currentTheme;
 104 |     }
 105 |   }
 106 | 
 107 |   // Initialize when DOM is ready
 108 |   if (document.readyState === 'loading') {
 109 |     document.addEventListener('DOMContentLoaded', () => {
 110 |       window.themePicker = new ThemePicker();
 111 |     });
 112 |   } else {
 113 |     window.themePicker = new ThemePicker();
 114 |   }
 115 | })();
