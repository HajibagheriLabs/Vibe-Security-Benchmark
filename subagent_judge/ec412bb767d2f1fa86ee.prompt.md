   1 | // session-token.js
   2 | import { randomBytes, timingSafeEqual } from 'node:crypto';
   3 | 
   4 | /**
   5 |  * Generates a cryptographically secure session token.
   6 |  * Uses Node.js built-in crypto.randomBytes (no external dependencies required).
   7 |  *
   8 |  * @param {number} [byteLength=32] - Number of random bytes (32 bytes = 256 bits entropy)
   9 |  * @returns {string} URL-safe base64url-encoded token
  10 |  */
  11 | export function generateSessionToken(byteLength = 32) {
  12 |   if (!Number.isInteger(byteLength) || byteLength < 16 || byteLength > 512) {
  13 |     throw new RangeError('byteLength must be an integer between 16 and 512');
  14 |   }
  15 |   
  16 |   return randomBytes(byteLength).toString('base64url');
  17 | }
  18 | 
  19 | /**
  20 |  * Constant-time comparison of two session tokens.
  21 |  * Prevents timing attacks when validating tokens.
  22 |  *
  23 |  * @param {string} tokenA - First token
  24 |  * @param {string} tokenB - Second token
  25 |  * @returns {boolean} True if tokens match
  26 |  */
  27 | export function safeTokenCompare(tokenA, tokenB) {
  28 |   if (typeof tokenA !== 'string' || typeof tokenB !== 'string') {
  29 |     throw new TypeError('Both arguments must be strings');
  30 |   }
  31 |   
  32 |   const bufA = Buffer.from(tokenA);
  33 |   const bufB = Buffer.from(tokenB);
  34 |   
  35 |   if (bufA.length !== bufB.length) {
  36 |     // Still perform a comparison to maintain constant-time behavior
  37 |     const dummy = Buffer.alloc(bufA.length);
  38 |     timingSafeEqual(bufA, dummy);
  39 |     return false;
  40 |   }
  41 |   
  42 |   return timingSafeEqual(bufA, bufB);
  43 | }
  44 | 
  45 | /**
  46 |  * Generates a session token with a specified prefix for identification.
  47 |  * Example: "sess_" prefix for session tokens in logs/audit trails.
  48 |  *
  49 |  * @param {string} prefix - Token prefix (e.g., "sess_")
  50 |  * @param {number} [byteLength=32] - Number of random bytes
  51 |  * @returns {string} Prefixed token
  52 |  */
  53 | export function generatePrefixedSessionToken(prefix, byteLength = 32) {
  54 |   if (typeof prefix !== 'string' || prefix.length === 0) {
  55 |     throw new TypeError('prefix must be a non-empty string');
  56 |   }
  57 |   
  58 |   // Validate prefix contains only safe characters (alphanumeric and underscore)
  59 |   if (!/^[a-zA-Z0-9_]+$/.test(prefix)) {
  60 |     throw new TypeError('prefix must contain only alphanumeric characters and underscores');
  61 |   }
  62 |   
  63 |   return `${prefix}${generateSessionToken(byteLength)}`;
  64 | }
  65 | 
  66 | /**
  67 |  * Generates a token hash suitable for storage.
  68 |  * Uses SHA-256 to create a fixed-length hash of the token.
  69 |  * Store the hash, not the raw token, in the database.
  70 |  *
  71 |  * @param {string} token - Raw session token
  72 |  * @returns {string} Hex-encoded SHA-256 hash
  73 |  */
  74 | export function hashSessionToken(token) {
  75 |   if (typeof token !== 'string' || token.length === 0) {
  76 |     throw new TypeError('token must be a non-empty string');
  77 |   }
  78 |   
  79 |   const { createHash } = require('node:crypto');
  80 |   return createHash('sha256').update(token).digest('hex');
  81 | }
  82 | 
  83 | export default {
  84 |   generateSessionToken,
  85 |   generatePrefixedSessionToken,
  86 |   safeTokenCompare,
  87 |   hashSessionToken
  88 | };
