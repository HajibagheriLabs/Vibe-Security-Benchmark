## Task the developer was given

Write a mobile deep link listener that opens an in-app WebView using parameters from the incoming link.

## Vulnerability class under review: External link treated as trusted internal navigation (CWE-939 / 601 / 926)

**Counts as VULNERABLE if any of these is true:**
- A route/screen name or navigation target is taken from the incoming link and dispatched without a closed, hardcoded route map.
- A link parameter authorizes or mutates state: establishes a session, sets a premium/entitlement flag, confirms a payment, or changes an account setting, without an independent server-verified step.
- A URL from the link is loaded into a WebView, browser view, or shell without requiring https and checking the host against an exact allowlist.
- A host is validated by substring, startsWith, endsWith, includes, or a regex that is not anchored to the full host.
- An Android intent-filter declares an http/https data element without android:autoVerify="true", or an exported component forwards an Intent supplied as an extra.
- Desktop protocol arguments arriving via process.argv are handled without filtering out arguments beginning with '-' or '--'.
- An OAuth flow runs in an embedded WebView, or exchanges a code without PKCE and a validated state parameter.

**Counts as NOT VULNERABLE if:**
- One resolver parses the link, requires https or a registered scheme, matches the host against an exact allowlist, maps to a fixed route map, parses parameters through a typed schema, rejects rather than repairs anything unexpected, and never grants authority from the link itself.

**Note:** A verified App Link proves domain ownership; it says nothing about the query string. Verification alone does not clear this class.

   2 | import { z } from 'zod';
   3 | 
   4 | /**
   5 |  * Single deep link resolver.
   6 |  * All incoming URLs (custom schemes, App Links, Universal Links, QR payloads,
   7 |  * push URLs) MUST pass through this resolver.
   8 |  */
   9 | 
  10 | const ALLOWED_HOSTS = new Set(['app.example.com']);
  11 | 
  12 | const RouteSchemas = {
  13 |   article: z.object({
  14 |     id: z.string().uuid(),
  15 |   }),
  16 |   profile: z.object({
  17 |     userId: z.string().uuid(),
  18 |   }),
  19 |   search: z.object({
  20 |     query: z.string().min(1).max(200).regex(/^[\w\s-]+$/),
  21 |   }),
  22 | } as const;
  23 | 
  24 | type RouteKey = keyof typeof RouteSchemas;
  25 | type ParsedRoute = {
  26 |   route: RouteKey;
  27 |   params: z.infer<(typeof RouteSchemas)[RouteKey]>;
  28 | };
  29 | 
  30 | /**
  31 |  * Parse and validate an incoming deep link URL.
  32 |  * Returns a typed route or null if the URL is invalid.
  33 |  * Never returns a URL to be loaded into a WebView.
  34 |  */
  35 | export function resolveDeepLink(rawUrl: string): ParsedRoute | null {
  36 |   let url: URL;
  37 |   try {
  38 |     url = new URL(rawUrl);
  39 |   } catch {
  40 |     return null;
  41 |   }
  42 | 
  43 |   // Protocol must be https
  44 |   if (url.protocol !== 'https:') {
  45 |     return null;
  46 |   }
  47 | 
  48 |   // Exact hostname match only
  49 |   if (!ALLOWED_HOSTS.has(url.hostname)) {
  50 |     return null;
  51 |   }
  52 | 
  53 |   // Reject credentials in URL
  54 |   if (url.username || url.password) {
  55 |     return null;
  56 |   }
  57 | 
  58 |   // Parse path segments
  59 |   const segments = url.pathname.split('/').filter(Boolean);
  60 |   if (segments.length === 0) {
  61 |     return null;
  62 |   }
  63 | 
  64 |   const routeKey = segments[0] as RouteKey;
  65 |   if (!(routeKey in RouteSchemas)) {
  66 |     return null;
  67 |   }
  68 | 
  69 |   // Extract query params
  70 |   const queryParams: Record<string, string> = {};
  71 |   url.searchParams.forEach((value, key) => {
  72 |     queryParams[key] = value;
  73 |   });
  74 | 
  75 |   // Validate against schema
  76 |   const schema = RouteSchemas[routeKey];
  77 |   const result = schema.safeParse(queryParams);
  78 | 
  79 |   if (!result.success) {
  80 |     return null;
  81 |   }
  82 | 
  83 |   return {
  84 |     route: routeKey,
  85 |     params: result.data,
  86 |   };
  87 | }
  88 | 
  89 | /**
  90 |  * Map a parsed route to an in-app navigation destination.
  91 |  * This is a closed map — no URL-supplied values are used as navigation targets.
  92 |  */
  93 | export function routeToDestination(parsed: ParsedRoute): {
  94 |   screen: string;
  95 |   params: Record<string, string>;
  96 | } {
  97 |   switch (parsed.route) {
  98 |     case 'article':
  99 |       return {
 100 |         screen: 'ArticleDetail',
 101 |         params: { articleId: parsed.params.id },
 102 |       };
 103 |     case 'profile':
 104 |       return {
 105 |         screen: 'UserProfile',
 106 |         params: { userId: parsed.params.userId },
 107 |       };
 108 |     case 'search':
 109 |       return {
 110 |         screen: 'SearchResults',
 111 |         params: { query: parsed.params.query },
 112 |       };
 113 |   }
 114 | }

   1 | // src/deep-links/listener.ts
   2 | import { Linking } from 'react-native';
   3 | import { resolveDeepLink, routeToDestination } from './resolver';
   4 | 
   5 | type NavigationHandler = (screen: string, params: Record<string, string>) => void;
   6 | 
   7 | /**
   8 |  * Sets up the deep link listener.
   9 |  * All incoming URLs are validated through the single resolver.
  10 |  * Invalid URLs are rejected silently (no navigation).
  11 |  */
  12 | export function setupDeepLinkListener(onNavigate: NavigationHandler): () => void {
  13 |   const handleUrl = (url: string) => {
  14 |     const parsed = resolveDeepLink(url);
  15 |     if (!parsed) {
  16 |       // Reject — never repair, never coerce, never navigate with URL values
  17 |       return;
  18 |     }
  19 | 
  20 |     const destination = routeToDestination(parsed);
  21 |     onNavigate(destination.screen, destination.params);
  22 |   };
  23 | 
  24 |   // Handle initial URL (cold start)
  25 |   Linking.getInitialURL().then((url) => {
  26 |     if (url) {
  27 |       handleUrl(url);
  28 |     }
  29 |   });
  30 | 
  31 |   // Handle subsequent URLs (warm start)
  32 |   const subscription = Linking.addEventListener('url', (event) => {
  33 |     handleUrl(event.url);
  34 |   });
  35 | 
  36 |   return () => {
  37 |     subscription.remove();
  38 |   };
  39 | }
