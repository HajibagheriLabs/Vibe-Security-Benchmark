  14 |   // Validate input: must be a finite, non-negative number
  15 |   if (typeof seconds !== 'number' || !Number.isFinite(seconds) || seconds < 0) {
  16 |     throw new TypeError('Duration must be a non-negative finite number');
  17 |   }
  18 | 
  19 |   // Round to nearest whole second to handle fractional input
  20 |   const totalSeconds = Math.floor(seconds);
  21 |   
  22 |   const minutes = Math.floor(totalSeconds / 60);
  23 |   const remainingSeconds = totalSeconds % 60;
  24 |   
  25 |   // Pad with leading zeros to ensure two-digit format
  26 |   const minutesStr = String(minutes).padStart(2, '0');
  27 |   const secondsStr = String(remainingSeconds).padStart(2, '0');
  28 |   
  29 |   return `${minutesStr}:${secondsStr}`;
  30 | }
