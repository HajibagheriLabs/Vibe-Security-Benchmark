   6 | const DEFAULT_THEME = 'light';

   8 | class ThemeManager {
   9 |   constructor() {
  10 |     this.currentTheme = DEFAULT_THEME;
  11 |     this.listeners = new Set();
  12 |     this.initialized = false;
  13 |   }
  14 | 
  15 |   /**
  16 |    * Initialize the theme manager
  17 |    * Loads saved preference or detects system preference
  18 |    */
  19 |   async init() {
  20 |     if (this.initialized) return;
  21 | 
  22 |     // Try to load from localStorage first
  23 |     const savedTheme = this.loadFromStorage();
  24 |     
  25 |     if (savedTheme) {
  26 |       this.currentTheme = savedTheme;
  27 |     } else {
  28 |       // Fall back to system preference
  29 |       this.currentTheme = this.getSystemTheme();
  30 |     }
  31 | 
  32 |     this.applyTheme(this.currentTheme);
  33 |     this.initialized = true;
  34 |     
  35 |     // Listen for system theme changes
  36 |     this.watchSystemTheme();
  37 |

  39 |   }

  41 |   /**
  42 |    * Load theme preference from localStorage
  43 |    * @returns {string|null} Saved theme or null
  44 |    */
  45 |   loadFromStorage() {
  46 |     try {
  47 |       const stored = localStorage.getItem(THEME_STORAGE_KEY);
  48 |       if (stored === 'light' || stored === 'dark') {
  49 |         return stored;
  50 |       }
  51 |     } catch (e) {
  52 |       console.warn('Failed to read theme from localStorage:', e);

  66 |     }

  85 | 
  86 |     const mediaQuery = window.matchMedia('(prefers-color-scheme: dark)');
  87 |     
  88 |     const handler = (e) => {
  89 |       // Only auto-switch if user hasn't explicitly set a preference
  90 |       const hasExplicitPreference = localStorage.getItem(THEME_STORAGE_KEY) !== null;
  91 |       if (!hasExplicitPreference) {
  92 |         this.setTheme(e.matches ? 'dark' : 'light');
  93 |       }
  94 |     };
  95 | 
  96 |     // Modern API
  97 |     if (mediaQuery.addEventListener) {
  98 |       mediaQuery.addEventListener('change', handler);

 100 |       // Legacy API
 101 |       mediaQuery.addListener(handler);
 102 |     }

 112 |   }

 176 |     }
 177 |     
 178 |     this.listeners.add(callback);

 180 |     // Return unsubscribe function

 183 |     };
 184 |   }
 185 | 
 186 |   /**

 195 |   }
 196 | }

 204 | } else {
 205 |   themeManager.init();
 206 | }
