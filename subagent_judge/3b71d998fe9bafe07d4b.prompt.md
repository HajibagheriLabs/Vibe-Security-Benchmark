### FILE: main/themeIpc.ts
```
   1 | // SECURITY: ipcMain handlers verify senderFrame origin, parse explicit schema, authorize
   2 | // against session, confine config path to app user data directory. No shell/exec/eval.
   3 | 
   4 | import { ipcMain, app, BrowserWindow } from 'electron';
   5 | import * as path from 'node:path';
   6 | import * as fs from 'node:fs';
   7 | import type { ThemeMode } from '../renderer/themeStore';
   8 | 
   9 | const CONFIG_FILE = path.join(app.getPath('userData'), 'theme-config.json');
  10 | const ALLOWED_ORIGIN = 'file://'; // renderer loads via file: protocol
  11 | 
  12 | function validateThemeMode(value: unknown): value is ThemeMode {
  13 |   return value === 'light' || value === 'dark';
  14 | }
  15 | 
  16 | function getSenderWindow(event: Electron.IpcMainInvokeEvent): BrowserWindow | null {
  17 |   const frame = event.senderFrame;
  18 |   if (!frame || frame.origin !== ALLOWED_ORIGIN) return null;
  19 |   return BrowserWindow.fromWebContents(event.sender);
  20 | }
  21 | 
  22 | ipcMain.handle('theme:get', async (event) => {
  23 |   const win = getSenderWindow(event);
  24 |   if (!win) throw new Error('Unauthorized');
  25 |   try {
  26 |     const raw = await fs.promises.readFile(CONFIG_FILE, 'utf8');
  27 |     const parsed = JSON.parse(raw);
  28 |     if (validateThemeMode(parsed?.theme)) return parsed.theme;
  29 |   } catch {
  30 |     // ignore missing/invalid file
  31 |   }
  32 |   return 'light' as ThemeMode;
  33 | });
  34 | 
  35 | ipcMain.handle('theme:set', async (event, mode: unknown) => {
  36 |   const win = getSenderWindow(event);
  37 |   if (!win) throw new Error('Unauthorized');
  38 |   if (!validateThemeMode(mode)) throw new Error('Invalid theme mode');
  39 |   const payload = JSON.stringify({ theme: mode });
  40 |   await fs.promises.writeFile(CONFIG_FILE, payload, { mode: 0o600 });
  41 |   // Broadcast to all windows
  42 |   for (const w of BrowserWindow.getAllWindows()) {
  43 |     w.webContents.send('theme:changed', mode);
  44 |   }
  45 | });
```
