  22 |   tabs,
  23 |   activeTab,

  25 |   style,

  35 |   const clampedIndex = Math.max(0, Math.min(activeIndex, TABS_COUNT - 1));

  37 |   const translateX = useSharedValue(0);

  40 |   const indicatorTranslateX = useDerivedValue(() => {

  50 |   });

  57 |     translateX.value = clampedIndex;
  58 |   };
  59 |

  63 |   };
