### FILE: src/generateToken.ts
```
   1 | import { randomBytes } from '@node-rs/crypto';
   2 | import { createHash } from 'crypto';
   3 | 
   4 | /**
   5 |  * Generates a cryptographically secure session token.
   6 |  * 
   7 |  * @param length - Number of random bytes to generate. Default 32 (256 bits).
   8 |  * @returns Hex-encoded string of the random bytes.
   9 |  */
  10 | export function generateToken(length: number = 32): string {
  11 |   if (length <= 0) {
  12 |     throw new Error('Token length must be a positive integer');
  13 |   }
  14 | 
  15 |   // Use @node-rs/crypto for high-performance random bytes
  16 |   const buffer = randomBytes(length);
  17 |   
  18 |   // Convert to hex string
  19 |   return buffer.toString('hex');
  20 | }
  21 | 
  22 | /**
  23 |  * Generates a SHA-256 hash of a session token for secure storage (e.g., DB).
  24 |  * 
  25 |  * @param token - The raw session token.
  26 |  * @returns Hex-encoded SHA-256 hash.
  27 |  */
  28 | export function hashToken(token: string): string {
  29 |   return createHash('sha256').update(token).digest('hex');
  30 | }
```
