   3 | /**
   4 |  * Formats a duration in seconds as `mm:ss`.
   5 |  *
   6 |  * Security note: This utility performs no I/O and handles no sensitive data.
   7 |  * Input is validated and coerced defensively; invalid input returns a safe
   8 |  * fallback string rather than throwing or producing NaN output.
   9 |  *
  10 |  * @param seconds - Duration in seconds (number or numeric string)
  11 |  * @returns Formatted string in `mm:ss` format, or "00:00" for invalid input
  12 |  */
  13 | export function formatDuration(seconds: number | string): string {
  14 |   // Parse and validate input defensively
  15 |   const numericSeconds = typeof seconds === 'string' ? Number(seconds) : seconds;
  16 | 
  17 |   if (!Number.isFinite(numericSeconds) || numericSeconds < 0) {
  18 |     return '00:00';
  19 |   }
  20 | 
  21 |   // Round to nearest whole second to avoid floating-point display artifacts
  22 |   const totalSeconds = Math.round(numericSeconds);
  23 | 
  24 |   const minutes = Math.floor(totalSeconds / 60);
  25 |   const remainingSeconds = totalSeconds % 60;
  26 | 
  27 |   const minutesStr = String(minutes).padStart(2, '0');
  28 |   const secondsStr = String(remainingSeconds).padStart(2, '0');
  29 | 
  30 |   return `${minutesStr}:${secondsStr}`;
  31 | }
