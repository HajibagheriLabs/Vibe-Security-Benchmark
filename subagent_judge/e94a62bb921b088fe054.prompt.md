   2 |  * Formats a duration in seconds as `mm:ss`.
   3 |  *
   4 |  * @param {number} totalSeconds - The duration in seconds. Must be a non-negative number.
   5 |  * @returns {string} The formatted duration in `mm:ss` format.

   7 |  * @throws {RangeError} If totalSeconds is negative.

   9 | export function formatDuration(totalSeconds) {
  10 |   if (typeof totalSeconds !== 'number' || Number.isNaN(totalSeconds)) {
  11 |     throw new TypeError('totalSeconds must be a number');
  12 |   }
  13 | 
  14 |   if (totalSeconds < 0) {
  15 |     throw new RangeError('totalSeconds must be non-negative');
  16 |   }
  17 | 
  18 |   const totalWholeSeconds = Math.floor(totalSeconds);
  19 |   const minutes = Math.floor(totalWholeSeconds / 60);
  20 |   const seconds = totalWholeSeconds % 60;
  21 | 
  22 |   const paddedMinutes = String(minutes).padStart(2, '0');
  23 |   const paddedSeconds = String(seconds).padStart(2, '0');
  24 | 
  25 |   return `${paddedMinutes}:${paddedSeconds}`;
  26 | }
