   1 | import { randomBytes, createHash, timingSafeEqual } from 'node:crypto';
   2 | 
   3 | /**
   4 |  * Generates a cryptographically secure session token.
   5 |  *
   6 |  * @param {number} [byteLength=32] - Number of random bytes (default 32 = 256 bits).
   7 |  * @param {'hex'|'base64'|'base64url'} [encoding='base64url'] - Output encoding.
   8 |  * @returns {string} A URL-safe, cryptographically secure token string.
   9 |  */
  10 | export function generateSessionToken(byteLength = 32, encoding = 'base64url') {
  11 |   if (!Number.isInteger(byteLength) || byteLength < 16) {
  12 |     throw new RangeError('byteLength must be an integer >= 16 (128 bits minimum)');
  13 |   }
  14 | 
  15 |   const validEncodings = ['hex', 'base64', 'base64url'];
  16 |   if (!validEncodings.includes(encoding)) {
  17 |     throw new TypeError(`encoding must be one of: ${validEncodings.join(', ')}`);
  18 |   }
  19 | 
  20 |   return randomBytes(byteLength).toString(encoding);
  21 | }
  22 | 
  23 | /**
  24 |  * Hashes a session token for storage using SHA-256.
  25 |  *
  26 |  * @param {string} token - The plaintext token to hash.
  27 |  * @returns {string} Hex-encoded SHA-256 hash of the token.
  28 |  */
  29 | export function hashSessionToken(token) {
  30 |   if (typeof token !== 'string' || token.length === 0) {
  31 |     throw new TypeError('token must be a non-empty string');
  32 |   }
  33 |   return createHash('sha256').update(token, 'utf8').digest('hex');
  34 | }
  35 | 
  36 | /**
  37 |  * Constant-time comparison of a plaintext token against a stored hash.
  38 |  *
  39 |  * @param {string} token - The plaintext token supplied by the client.
  40 |  * @param {string} storedHash - The hex-encoded SHA-256 hash stored server-side.
  41 |  * @returns {boolean} True if the token matches the hash.
  42 |  */
  43 | export function verifySessionToken(token, storedHash) {
  44 |   if (typeof token !== 'string' || typeof storedHash !== 'string') {
  45 |     return false;
  46 |   }
  47 | 
  48 |   const computedHash = hashSessionToken(token);
  49 |   const computedBuffer = Buffer.from(computedHash, 'hex');
  50 |   const storedBuffer = Buffer.from(storedHash, 'hex');
  51 | 
  52 |   if (computedBuffer.length !== storedBuffer.length) {
  53 |     return false;
  54 |   }
  55 | 
  56 |   return timingSafeEqual(computedBuffer, storedBuffer);
  57 | }
  58 | 
  59 | /**
  60 |  * Generates a token along with its hash in one call.
  61 |  *
  62 |  * @param {number} [byteLength=32] - Number of random bytes.
  63 |  * @returns {{ token: string, hash: string }} The plaintext token and its SHA-256 hash.
  64 |  */
  65 | export function generateTokenPair(byteLength = 32) {
  66 |   const token = generateSessionToken(byteLength);
  67 |   const hash = hashSessionToken(token);
  68 |   return { token, hash };
  69 | }
